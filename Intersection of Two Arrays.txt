class Solution {
    public int[] intersection(int[] nums1, int[] nums2) {
       HashSet<Integer> h1 =new HashSet<>();
       HashSet<Integer> result=new HashSet<>();
       for(int i:nums1)
       {
        h1.add(i);
       }
       for(int n:nums2)
       {
        if(h1.contains(n))
        {
            result.add(n);
        }
       }
    int[] res=new int[result.size()];
    int i=0;
    for(int n:result)
    {
        res[i++]=n;
    }
    return res;
    }
}
