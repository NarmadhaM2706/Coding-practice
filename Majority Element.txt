class Solution {
    public int majorityElement(int[] nums) {
    Map<Integer,Integer> map= new HashMap<>();
    for(int i:nums)
    {
            map.put(i,map.getOrDefault(i,0)+1);
    }
    int max=0;
    int freq=0;
    for(int i:nums)
    {
        if(map.get(i)>freq){
         freq=map.get(i);
         max=i;
        }
    }
    return max;
    }
}
